# ВНИМАНИЕ: пакеты ниже покрывают только Python-слой. Системные бинарники
# (ffmpeg, xdg-desktop-portal, grim/slurp, scrot) ставятся пакетным
# менеджером дистрибутива, см. блок "Системные зависимости" в конце файла.

# Базовый GUI-фреймворк: окна, трей, QGraphicsView-редактор, QProcess.
PySide6>=6.6.0

# Асинхронный чистокровный Python-клиент D-Bus без GLib-петли.
# Используется для org.freedesktop.portal.ScreenCast и GlobalShortcuts.
dbus-next>=0.2.3

# Перехват глобальных хоткеев в сессии X11 (в Wayland не применяется).
pynput>=1.7.6

# Низкоуровневые запросы к X-серверу: список мониторов, геометрия и
# определение активного окна для режима "снимок активного окна".
python-xlib>=0.33; sys_platform == "linux"

# Обработка растровых изображений: конвертация буферов захвата,
# сохранение PNG/JPEG/WEBP/BMP с тонкой настройкой параметров сжатия.
Pillow>=10.2.0

# Плагин Pillow для записи AVIF (отдельный пакет с libavif внутри колеса).
pillow-avif-plugin>=1.4.3

# Системные зависимости (устанавливаются вне pip):
#   ffmpeg                     - кодирование видео, GIF, WebP, APNG
#   xdg-desktop-portal         - порталы ScreenCast / GlobalShortcuts
#   xdg-desktop-portal-gtk|-kde|-wlr|-hyprland - бэкенд портала под DE
#   pipewire, pipewire-pulse   - транспорт видеопотока и аудиоисточников
#   grim, slurp                - фоллбэк-скриншоты и выделение в wlroots
#   scrot | maim               - фоллбэк-скриншоты в X11
#   wl-clipboard | xclip       - фоллбэк копирования в буфер обмена
#   libxcb-cursor0             - обязателен для платформенного модуля Qt xcb
# --------------------------------------------------------------------------
